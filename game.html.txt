        // ---------- GAME STATE ----------

        // ---------- DOM REFS ----------

        const playersGrid = $('playersGrid');

        // ---------- UTILITY ----------
        function shuffle(arr) {
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }
        
        function getAlive() { return game.players.filter(p => p.isAlive); }
        function getAliveGood() { return game.players.filter(p => p.isAlive && p.team === 'good'); }
        function getAliveEvil() { return game.players.filter(p => p.isAlive && p.team === 'evil'); }
        function getRoleFile(roleName) {
            const found = ROLES_LIST.find(r => r.name === roleName);
            return found ? found.file : 'villager.png';
        }
        function getRoleEmoji(roleName) {
            const found = ROLES_LIST.find(r => r.name === roleName);
            return found ? found.emoji : '❓';
        }
        function getRoleTeam(roleName) {
            const found = ROLES_LIST.find(r => r.name === roleName);
            return found ? found.team : 'good';
        }
        
        // ---------- INIT GAME ----------
        function initGame() {
            const playerCount = 10; // می‌توانید تغییر دهید
            const names = ['سارا', 'مهدی', 'نیلوفر', 'امیر', 'زهرا', 'رضا', 'مینا', 'کیان', 'سپهر', 'آتنا'];
            
            // انتخاب نقش‌ها بر اساس تعداد بازیکنان
            let selectedRoles = [];
            const allGood = ROLES_LIST.filter(r => r.team === 'good');
            const allEvil = ROLES_LIST.filter(r => r.team === 'evil');
            
            // نقش‌های کلیدی
            const coreGood = ['روستایی', 'دکتر', 'پیشگو'];
            const coreEvil = ['گرگینه', 'گرگینه'];
            
            selectedRoles = [...coreGood, ...coreEvil];
            
            // اضافه کردن نقش‌های بیشتر
            const extraCount = playerCount - selectedRoles.length;
            if (extraCount > 0) {
                const extraGood = allGood.filter(r => !coreGood.includes(r.name) && r.name !== 'روستایی');
                const extraEvil = allEvil.filter(r => !coreEvil.includes(r.name));
                const shuffledGood = shuffle([...extraGood]);
                const shuffledEvil = shuffle([...extraEvil]);
                
                const goodCount = Math.ceil(extraCount * 0.6);
                const evilCount = Math.floor(extraCount * 0.4);
                
                for (let i = 0; i < goodCount && i < shuffledGood.length; i++) {
                    selectedRoles.push(shuffledGood[i].name);
                }
                for (let i = 0; i < evilCount && i < shuffledEvil.length; i++) {
                    selectedRoles.push(shuffledEvil[i].name);
                }
            }
            
            // پر کردن باقی‌مانده با روستایی
            while (selectedRoles.length < playerCount) {
                selectedRoles.push('روستایی');
            }
            
            shuffle(selectedRoles);
            
            // ساخت بازیکنان
            game.players = [];
            for (let i = 0; i < playerCount; i++) {
                const role = selectedRoles[i] || 'روستایی';
                game.players.push({
                    id: i,
                    name: names[i] || `بازیکن ${i+1}`,
                    role: role,
                    team: getRoleTeam(role),
                    isAlive: true,
                    isRevealed: false
                });
            }
            
            game.phase = 'idle';
            game.round = 0;
            game.winner = null;
            game.log = [];
            
            addLog(`🎮 بازی جدید با ${playerCount} بازیکن شروع شد!`);
            const roleList = game.players.map(p => p.role).join('، ');
            addLog(`📋 نقش‌ها: ${roleList}`);
            
            render();
            updateUI();
        }
        
        // ---------- LOGGING ----------
        function addLog(message) {
            const time = new Date().toLocaleTimeString('fa-IR');
            game.log.push({ time, message });
            renderLog();
        }
        
        function renderLog() {
            logBox.innerHTML = game.log.slice(-20).map(entry =>
                `<div class="log-entry">
                    <span class="time">[${entry.time}]</span>
                    ${entry.message}
                </div>`
            ).join('');
            logBox.scrollTop = logBox.scrollHeight;
        }
        
        // ---------- GAME LOGIC ----------
        function startGame() {
            if (game.phase !== 'idle') return;
            game.phase = 'night';
            game.round = 1;
            addLog('🌙 شب فرا رسید... گرگ‌ها بیدار می‌شوند!');
            
            // شبیه‌سازی خودکار
            setTimeout(() => {
                if (game.phase === 'night') runNightPhase();
            }, 1500);
            
            render();
            updateUI();
        }
        
        function runNightPhase() {
            const alive = getAlive();
            const aliveGood = getAliveGood();
            const aliveEvil = getAliveEvil();
            
            if (checkWinner()) return;
            
            // گرگ‌ها یک نفر را می‌کشند
            const wolves = aliveEvil.filter(p => p.role === 'گرگینه' || p.role === 'گرگ آلفا');
            if (wolves.length > 0 && aliveGood.length > 0) {
                const target = aliveGood[Math.floor(Math.random() * aliveGood.length)];
                if (target) {
                    target.isAlive = false;
                    addLog(`🐺 گرگ‌ها ${target.name} (${target.role}) را کشتند!`);
                }
            }
            
            // دکتر نجات می‌دهد (شانس ۶۰٪)
            const doctors = aliveGood.filter(p => p.role === 'دکتر');
            if (doctors.length > 0 && Math.random() > 0.4) {
                const victim = game.players.find(p => !p.isAlive && p.team === 'good');
                if (victim) {
                    victim.isAlive = true;
                    addLog(`💉 دکتر ${victim.name} را نجات داد!`);
                }
            }
            
            // پیشگو بررسی می‌کند
            const seers = aliveGood.filter(p => p.role === 'پیشگو');
            if (seers.length > 0) {
                const target = alive[Math.floor(Math.random() * alive.length)];
                if (target) {
                    const teamLabel = target.team === 'evil' ? 'شر' : 'خیر';
                    addLog(`🔮 پیشگو ${target.name} را بررسی کرد: ${target.role} (${teamLabel})`);
                }
            }
            
            if (checkWinner()) return;
            
            // رفتن به روز
            game.phase = 'day';
            addLog(`☀️ روز ${game.round} فرا رسید. رأی‌گیری برای اعدام.`);
            
            // رأی‌گیری خودکار
            setTimeout(() => {
                if (game.phase === 'day') runDayPhase();
            }, 2000);
            
            render();
            updateUI();
        }
        
        function runDayPhase() {
            const alive = getAlive();
            if (checkWinner()) return;
            
            // رأی‌گیری خودکار
            if (alive.length > 1) {
                const accused = alive[Math.floor(Math.random() * alive.length)];
                if (accused) {
                    // چک کردن شاهدخت
                    if (accused.role === 'شاهدخت' && !accused.isRevealed) {
                        accused.isRevealed = true;
                        addLog(`👸 شاهدخت ${accused.name} از اعدام جان سالم به در برد!`);
                    } else {
                        accused.isAlive = false;
                        addLog(`⚖️ ${accused.name} (${accused.role}) اعدام شد!`);
                        
                        // چک کردن دلقک
                        if (accused.role === 'دلقک') {
                            game.winner = 'jester';
                            addLog(`🤡 دلقک ${accused.name} برنده شد!`);
                            endGame();
                            return;
                        }
                    }
                }
            }
            
            if (checkWinner()) return;
            
            // رفتن به شب
            game.phase = 'night';
            game.round++;
            addLog(`🌙 شب ${game.round} فرا رسید...`);
            
            setTimeout(() => {
                if (game.phase === 'night') runNightPhase();
            }, 1500);
            
            render();
            updateUI();
        }
        
        function checkWinner() {
            const aliveGood = getAliveGood();
            const aliveEvil = getAliveEvil();
            
            if (aliveEvil.length === 0) {
                game.winner = 'good';
                addLog('🎉 تیم خیر (خوب‌ها) برنده شدند!');
                endGame();
                return true;
            }
            if (aliveGood.length === 0 || aliveGood.length <= aliveEvil.length) {
                game.winner = 'evil';
                addLog('🐺 تیم شر (گرگ‌ها) برنده شدند!');
                endGame();
                return true;
            }
            return false;
        }
        
        function endGame() {
            game.phase = 'ended';
            game.players.forEach(p => p.isRevealed = true);
            render();
            updateUI();
            
            if (game.winner === 'good') {
                winnerDisplay.innerHTML = `<div class="winner-banner">🎉 برنده: تیم خیر (خوب‌ها)</div>`;
            } else if (game.winner === 'evil') {
                winnerDisplay.innerHTML = `<div class="winner-banner">🐺 برنده: تیم شر (گرگ‌ها)</div>`;
            } else if (game.winner === 'jester') {
                winnerDisplay.innerHTML = `<div class="winner-banner">🤡 برنده: دلقک!</div>`;
            }
        }
        
        function nextPhase() {
            if (game.phase === 'night') {
                runNightPhase();
            } else if (game.phase === 'day') {
                runDayPhase();
            }
        }
        
        function resetGame() {
            winnerDisplay.innerHTML = '';
            if (game.phase === 'ended') {
                initGame();
            } else {
                initGame();
            }
        }
        
        // ---------- RENDER ----------
        function render() {
            const all = game.players;
            
            playersGrid.innerHTML = all.map(p => {
                const isDead = !p.isAlive;
                const isRevealed = p.isRevealed || game.phase === 'ended';
                const teamClass = p.team === 'good' ? 'role-good' : 'role-evil';
                const roleDisplay = isRevealed ? `${getRoleEmoji(p.role)} ${p.role}` : '❓';
                const imgSrc = isRevealed ? `cards/${getRoleFile(p.role)}` : 'cards/hidden.png';
                const imgClass = isDead ? 'card-image dead' : (isRevealed ? 'card-image' : 'card-image hidden');
                
                return `
                    <div class="player-card ${isDead ? 'dead' : ''}">
                        <img src="${imgSrc}" alt="${p.role}" class="${imgClass}"
                             onerror="this.src='data:image/svg+xml,%3Csvg xmlns=%22http://www.w3.org/2000/svg%22 width=%22120%22 height=%22160%22%3E%3Crect width=%22120%22 height=%22160%22 fill=%22%23333%22/%3E%3Ctext x=%2260%22 y=%2280%22 text-anchor=%22middle%22 fill=%22%23666%22 font-size=%2230%22%3E${getRoleEmoji(p.role)}%3C/text%3E%3C/svg%3E'">
                        <div class="name">${p.name} ${p.team === 'evil' ? '🐺' : '👼'}</div>
                        <div class="role-text ${isRevealed ? teamClass : 'role-hidden'}">${roleDisplay}</div>
                        <div class="alive-status">${isDead ? '💀 مرده' : '❤️ زنده'}</div>
                    </div>
                `;
            }).join('');
            
            // Update stats
            aliveCount.textContent = getAlive().length;
            goodCount.textContent = getAliveGood().length;
            evilCount.textContent = getAliveEvil().length;
            
            // Phase badge
            let phaseText = '';
            let phaseClass = '';
            switch (game.phase) {
                case 'idle': phaseText = '⏳ آماده'; phaseClass = 'phase-idle'; break;
                case 'night': phaseText = `🌙 شب ${game.round}`; phaseClass = 'phase-night'; break;
                case 'day': phaseText = `☀️ روز ${game.round}`; phaseClass = 'phase-day'; break;
                case 'ended': phaseText = '🏁 پایان'; phaseClass = 'phase-ended'; break;
                default: phaseText = game.phase; phaseClass = 'phase-idle';
            }
            phaseBadge.textContent = phaseText;
            phaseBadge.className = `phase-badge ${phaseClass}`;
            roundDisplay.textContent = game.round;
        }
        
        function updateUI() {
            const isIdle = game.phase === 'idle';
            const isEnded = game.phase === 'ended';
            
            btnStart.disabled = !isIdle;
            btnNext.disabled = isIdle || isEnded;
        }
        
        // ---------- INIT ----------
        initGame();
    </script>
